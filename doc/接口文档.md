# 数据库建表语句

```sql
CREATE TABLE `item`(
    `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `cover_images` varchar(2000) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '封面图，$拼接',
    `name` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '商品名字',
    `price` decimal(10,2) NOT NULL DEFAULT 0.00 COMMENT '商品价格，单位元',
    `introduction` varchar(2000) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '商品介绍',
    `create_time` int unsigned NOT NULL COMMENT '创建时间',
    `update_time` int unsigned NOT NULL COMMENT '修改时间',
    `is_deleted` tinyint unsigned NOT NULL DEFAULT 0 COMMENT '是否已经被删除 0未删除，1已删除',
    PRIMARY KEY (`id`),
    KEY `idx_name` (`name`) 
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT '商品表'
```

修改item表

```mysql
-- 第一步：新增，允许null
ALTER TABLE item ADD COLUMN category_id bigint UNSIGNED COMMENT '商品分类ID' AFTER is_deleted;

-- 第二步：执行之前的update语句，给所有商品填充category_id
-- UPDATE item SET category_id = ? WHERE ...;

-- 第三步：全部数据填充完成后，再修改为NOT NULL
ALTER TABLE item MODIFY COLUMN category_id INT UNSIGNED NOT NULL COMMENT '商品分类ID';
```

category表：

```mysql
create table category
(
    id          bigint unsigned primary key auto_increment comment '商品分类ID',
    name        varchar(50)                not null comment '商品分类名',
    image       varchar(100)               not null comment '商品分类图片',
    create_time int unsigned               not null comment '创建时间',
    update_time int unsigned               not null comment '修改时间',
    is_deleted  tinyint unsigned default 0 not null comment '是否已经被删除 0未删除，1已删除'
) ENGINE = InnoDB
  DEFAULT CHARSET = utf8mb4
  COLLATE = utf8mb4_unicode_ci comment '商品分类表';
```

file表:

```sql
create table file
(
    id            bigint unsigned primary key auto_increment comment '主键ID',
    type          tinyint unsigned             not null comment '文件类型：1图片 2视频 3文件',
    original_name varchar(255)                 not null comment '文件原始名',
    object_name   varchar(255)                 not null comment 'OSS对象名/路径',
    url           varchar(1024)                not null comment 'OSS访问url',
    create_time   int unsigned                 not null comment '创建时间',
    update_time   int unsigned                 not null comment '修改时间',
    is_deleted    tinyint unsigned default '0' not null comment '是否已经被删除 0未删除，1已删除'
) ENGINE = InnoDB
  DEFAULT CHARSET = utf8mb4
  COLLATE = utf8mb4_unicode_ci comment '文件类型表';
```
